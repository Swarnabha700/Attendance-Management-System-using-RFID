CREATE USER C##attendance IDENTIFIED BY 123;
GRANT DBA TO C##attendance;

CREATE TABLE cards (
    id VARCHAR2(20) PRIMARY KEY,
    status VARCHAR2(20)
);

CREATE TABLE students (
    roll VARCHAR2(20) PRIMARY KEY,
    fname VARCHAR2(50),
    lname VARCHAR2(50),
    department VARCHAR2(100),
    scode VARCHAR2(10),
    card_id VARCHAR2(20),
    CONSTRAINT fk_card
        FOREIGN KEY (card_id) REFERENCES cards(id)
);

CREATE TABLE attendance (
    roll VARCHAR2(20),
    fname VARCHAR2(50),
    lname VARCHAR2(50),
    department VARCHAR2(100),
    scode VARCHAR2(10),
    attendance_time TIMESTAMP,
    CONSTRAINT fk_student FOREIGN KEY (roll) REFERENCES students(roll)
);
